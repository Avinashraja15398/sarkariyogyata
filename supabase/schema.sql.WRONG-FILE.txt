import { notFound } from 'next/navigation';
import Link from 'next/link';
import { pub, todayIST } from '@/lib/public';
import EligibilityCalculator from './EligibilityCalculator';
import PhotoResizer from './PhotoResizer';

export const revalidate = 300;

interface JobPageProps {
  params: Promise<{ slug: string }>;
}

export default async function JobDetail({ params }: JobPageProps) {
  const { slug } = await params;
  const supabase = pub();

  const { data: job, error } = await supabase
    .from('jobs')
    .select('*')
    .eq('slug', slug)
    .single();

  if (error || !job) {
    notFound();
  }

  const today = todayIST();
  const isClosed = job.last_date ? job.last_date < today : false;

  return (
    <div className="min-h-screen bg-slate-50 text-slate-800 antialiased pb-16">
      {/* Top Header / Breadcrumb */}
      <div className="border-b border-slate-200 bg-white">
        <div className="mx-auto max-w-6xl px-4 py-3">
          <nav className="flex items-center space-x-2 text-xs font-medium text-slate-500">
            <Link href="/" className="hover:text-blue-600 transition-colors">Home</Link>
            <span>/</span>
            <span className="capitalize text-slate-600">{job.category || 'Govt'} Jobs</span>
            <span>/</span>
            <span className="truncate text-slate-400 max-w-[200px] md:max-w-md">{job.short_title || job.title}</span>
          </nav>
        </div>
      </div>

      <main className="mx-auto max-w-6xl px-4 pt-6">
        {/* Hero Section */}
        <div className="rounded-2xl border border-slate-200 bg-white p-6 md:p-8 shadow-sm">
          <div className="flex flex-wrap items-center gap-2 mb-3">
            <span className="inline-flex items-center rounded-md bg-blue-50 px-2.5 py-1 text-xs font-semibold text-blue-700 ring-1 ring-inset ring-blue-600/20">
              {job.organization}
            </span>
            <span className="inline-flex items-center rounded-md bg-slate-100 px-2.5 py-1 text-xs font-medium text-slate-600">
              {job.category || 'Govt'}
            </span>
            {isClosed ? (
              <span className="inline-flex items-center rounded-md bg-rose-50 px-2.5 py-1 text-xs font-semibold text-rose-700 ring-1 ring-inset ring-rose-600/20">
                ● Application Closed
              </span>
            ) : (
              <span className="inline-flex items-center rounded-md bg-emerald-50 px-2.5 py-1 text-xs font-semibold text-emerald-700 ring-1 ring-inset ring-emerald-600/20 animate-pulse">
                ● Active Form
              </span>
            )}
          </div>

          <h1 className="text-2xl md:text-3xl font-extrabold tracking-tight text-slate-900 leading-snug">
            {job.title}
          </h1>

          {job.brief_info && (
            <p className="mt-3 text-sm md:text-base text-slate-600 leading-relaxed border-l-4 border-blue-500 pl-3">
              {job.brief_info}
            </p>
          )}

          {/* Key Metrics Quick Bar */}
          <div className="mt-6 grid grid-cols-2 gap-3 sm:grid-cols-4">
            <div className="rounded-xl bg-slate-50 p-4 border border-slate-100">
              <span className="text-xs font-medium text-slate-500">कुल पद (Posts)</span>
              <p className="mt-1 text-xl font-bold text-slate-900">{job.total_posts ? job.total_posts.toLocaleString('en-IN') : 'N/A'}</p>
            </div>
            <div className="rounded-xl bg-slate-50 p-4 border border-slate-100">
              <span className="text-xs font-medium text-slate-500">आवेदन की अंतिम तिथि</span>
              <p className={`mt-1 text-lg font-bold ${isClosed ? 'text-rose-600' : 'text-blue-700'}`}>
                {job.last_date || 'N/A'}
              </p>
            </div>
            <div className="rounded-xl bg-slate-50 p-4 border border-slate-100">
              <span className="text-xs font-medium text-slate-500">उम्र सीमा</span>
              <p className="mt-1 text-lg font-bold text-slate-900">
                {job.min_age && job.max_age ? `${job.min_age} - ${job.max_age} Years` : 'Check Notice'}
              </p>
            </div>
            <div className="rounded-xl bg-slate-50 p-4 border border-slate-100">
              <span className="text-xs font-medium text-slate-500">योग्यता</span>
              <p className="mt-1 text-sm font-semibold text-slate-800 line-clamp-2">
                {job.eligibility_summary || '10th / 12th / Degree'}
              </p>
            </div>
          </div>
        </div>

        {/* 2-Column Responsive Body */}
        <div className="mt-8 grid grid-cols-1 gap-8 lg:grid-cols-3">
          {/* Left Column (Details) - 2 Cols Wide */}
          <div className="space-y-6 lg:col-span-2">
            {/* Dates & Fees Grid */}
            <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
              {/* Important Dates */}
              <div className="rounded-2xl border border-slate-200 bg-white p-6 shadow-sm">
                <h2 className="text-base font-bold text-slate-900 flex items-center gap-2 pb-3 border-b border-slate-100">
                  <span className="text-blue-600">📅</span> महत्वपूर्ण तिथियां (Dates)
                </h2>
                <ul className="mt-4 space-y-3 text-sm">
                  <li className="flex justify-between items-center">
                    <span className="text-slate-500">आवेदन शुरू:</span>
                    <span className="font-semibold text-slate-800">{job.application_start || 'Notified Soon'}</span>
                  </li>
                  <li className="flex justify-between items-center">
                    <span className="text-slate-500">अंतिम तिथि:</span>
                    <span className="font-semibold text-rose-600">{job.last_date || 'Notified Soon'}</span>
                  </li>
                  <li className="flex justify-between items-center">
                    <span className="text-slate-500">फीस भरने की अंतिम तिथि:</span>
                    <span className="font-semibold text-slate-800">{job.fee_last_date || job.last_date || 'N/A'}</span>
                  </li>
                  <li className="flex justify-between items-center">
                    <span className="text-slate-500">परीक्षा तिथि:</span>
                    <span className="font-semibold text-blue-700">{job.exam_date || 'As per schedule'}</span>
                  </li>
                </ul>
              </div>

              {/* Application Fee */}
              <div className="rounded-2xl border border-slate-200 bg-white p-6 shadow-sm">
                <h2 className="text-base font-bold text-slate-900 flex items-center gap-2 pb-3 border-b border-slate-100">
                  <span className="text-amber-500">💳</span> आवेदन शुल्क (Fee)
                </h2>
                <ul className="mt-4 space-y-3 text-sm">
                  <li className="flex justify-between items-center">
                    <span className="text-slate-500">GEN / OBC / EWS:</span>
                    <span className="font-bold text-slate-900">
                      {job.fee_gen_obc_ews !== null ? `₹${job.fee_gen_obc_ews}` : '₹0'}
                    </span>
                  </li>
                  <li className="flex justify-between items-center">
                    <span className="text-slate-500">SC / ST / PH:</span>
                    <span className="font-bold text-slate-900">
                      {job.fee_sc_st_ph !== null ? `₹${job.fee_sc_st_ph}` : '₹0'}
                    </span>
                  </li>
                  <li className="flex justify-between items-center">
                    <span className="text-slate-500">सभी महिला उम्मीदवार:</span>
                    <span className="font-bold text-slate-900">
                      {job.fee_female !== null ? `₹${job.fee_female}` : '₹0'}
                    </span>
                  </li>
                  <li className="pt-2 text-xs text-slate-400 border-t border-slate-50">
                    भुगतान केवल ऑनलाइन माध्यम (Debit / Credit / Net Banking / UPI) से मान्य है।
                  </li>
                </ul>
              </div>
            </div>

            {/* "Meri Yogyata" Interactive Age Calculator */}
            <div className="rounded-2xl border border-blue-100 bg-blue-50/40 p-6 shadow-sm">
              <div className="flex items-center gap-2">
                <span className="flex h-8 w-8 items-center justify-center rounded-lg bg-blue-600 text-white font-bold text-sm">✓</span>
                <div>
                  <h2 className="text-base font-bold text-slate-900">मेरी योग्यता (Age Eligibility Calculator)</h2>
                  <p className="text-xs text-slate-500">अपनी जन्मतिथि दर्ज करके तुरंत जानें कि आप आवेदन कर सकते हैं या नहीं</p>
                </div>
              </div>
              <div className="mt-4">
                <EligibilityCalculator
                  minAge={job.min_age || 18}
                  maxAge={job.max_age || 35}
                  ageAsOn={job.age_as_on || job.last_date || '2026-07-01'}
                />
              </div>
            </div>

            {/* Photo Resizer Tool */}
            <div className="rounded-2xl border border-slate-200 bg-white p-6 shadow-sm">
              <div className="flex items-center gap-2">
                <span className="text-xl">📷</span>
                <div>
                  <h2 className="text-base font-bold text-slate-900">Photo & Signature Resizer (20 - 50 KB)</h2>
                  <p className="text-xs text-slate-500">100% प्राइवेट: फोटो आपके ब्राउज़र में ही प्रोसेस होती है, कहीं अपलोड नहीं होती।</p>
                </div>
              </div>
              <div className="mt-4">
                <PhotoResizer />
              </div>
            </div>

            {/* Official Updates Timeline */}
            <div className="rounded-2xl border border-slate-200 bg-white p-6 shadow-sm">
              <h2 className="text-base font-bold text-slate-900 flex items-center gap-2 pb-3 border-b border-slate-100">
                <span className="text-purple-600">🔔</span> अपडेट्स और संशोधन (Audit Log)
              </h2>
              <div className="mt-4 space-y-3">
                <div className="flex items-start gap-3 text-sm">
                  <div className="h-2 w-2 rounded-full bg-emerald-500 mt-1.5 shrink-0" />
                  <div>
                    <span className="font-semibold text-slate-800">New Notification Released</span>
                    <p className="text-xs text-slate-400">विस्तृत अधिसूचना विभाग द्वारा आधिकारिक वेबसाइट पर जारी कर दी गई है।</p>
                  </div>
                </div>
              </div>
            </div>
          </div>

          {/* Right Column (Sticky Quick Action Bar) - 1 Col Wide */}
          <div className="space-y-6">
            <div className="sticky top-6 rounded-2xl border border-slate-200 bg-white p-6 shadow-sm">
              <h2 className="text-sm font-bold uppercase tracking-wider text-slate-400">महत्वपूर्ण लिंक्स</h2>
              
              <div className="mt-4 space-y-3">
                {job.apply_online_url ? (
                  <a
                    href={job.apply_online_url}
                    target="_blank"
                    rel="noopener noreferrer"
                    className={`flex w-full items-center justify-center rounded-xl px-4 py-3.5 text-center text-sm font-bold text-white shadow-sm transition-all ${
                      isClosed
                        ? 'bg-slate-400 cursor-not-allowed opacity-75'
                        : 'bg-emerald-600 hover:bg-emerald-700 hover:shadow-emerald-600/20 shadow-md'
                    }`}
                  >
                    {isClosed ? 'आवेदन बंद हो चुका है' : 'ऑनलाइन आवेदन करें (Apply Online) →'}
                  </a>
                ) : (
                  <button disabled className="w-full rounded-xl bg-slate-200 px-4 py-3 text-sm font-semibold text-slate-500 cursor-not-allowed">
                    Link Activate Soon
                  </button>
                )}

                {job.notification_url && (
                  <a
                    href={job.notification_url}
                    target="_blank"
                    rel="noopener noreferrer"
                    className="flex w-full items-center justify-center rounded-xl border border-slate-200 bg-white px-4 py-3 text-sm font-semibold text-slate-700 hover:bg-slate-50 hover:border-slate-300 transition-colors"
                  >
                    📄 डाउनलोड नोटिफिकेशन (PDF)
                  </a>
                )}

                {job.official_website && (
                  <a
                    href={job.official_website}
                    target="_blank"
                    rel="noopener noreferrer"
                    className="flex w-full items-center justify-center rounded-xl border border-slate-200 bg-white px-4 py-2.5 text-xs font-medium text-slate-600 hover:bg-slate-50 transition-colors"
                  >
                    🌐 आधिकारिक वेबसाइट देखें
                  </a>
                )}
              </div>

              <hr className="my-5 border-slate-100" />

              {/* Share on WhatsApp */}
              <a
                href={`https://api.whatsapp.com/send?text=${encodeURIComponent(
                  `*${job.title}*\nकुल पद: ${job.total_posts || 'Check Notice'}\nअंतिम तिथि: ${job.last_date \vert{}\vert{} ''}\nपूरी जानकारी और फॉर्म लिंक यहाँ देखें:\nhttps://sarkariyogyata.com/jobs/${job.slug}`
                )}`}
                target="_blank"
                rel="noopener noreferrer"
                className="flex w-full items-center justify-center gap-2 rounded-xl bg-emerald-50 px-4 py-3 text-xs font-bold text-emerald-800 hover:bg-emerald-100 transition-colors"
              >
                <span>💬</span> दोस्तों के साथ WhatsApp पर शेयर करें
              </a>

              <div className="mt-4 rounded-lg bg-amber-50 p-3 text-[11px] text-amber-800 leading-relaxed border border-amber-200/60">
                <strong>नोट:</strong> आवेदन करने से पहले विभाग का आधिकारिक नोटिफिकेशन अवश्य पढ़ें।
              </div>
            </div>
          </div>
        </div>
      </main>
    </div>
  );
}
